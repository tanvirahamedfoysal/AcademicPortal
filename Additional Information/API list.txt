======================================================
1. AUTHENTICATION & ACCOUNT API
======================================================
POST    /api/v1/auth/login                           # Authenticate user and return JWT (Public)
GET     /api/v1/auth/validate-username               # Check username availability/validity (Public)
POST    /api/v1/auth/register/request-otp            # Send OTP for student registration (Public)
POST    /api/v1/auth/register                        # Register a new student (initial status: PENDING) (Public)
POST    /api/v1/auth/password-reset/request-otp      # Send OTP for password reset (Public)
PATCH   /api/v1/auth/password-reset                  # Verify OTP and reset password (Public)


======================================================
2. PROFILE API
======================================================
GET     /api/v1/profile/me                           # Get current user's profile (Authenticated)
PATCH   /api/v1/profile/me                           # Update current user's profile (Authenticated)
# Note: Editable fields depend on the user's role (Admin/Moderator/Student).


======================================================
3. STUDENT MANAGEMENT API
======================================================
GET     /api/v1/students                            # List verified students/alumni (Public)
GET     /api/v1/students/:id                        # Get public profile of a verified student (Public)

PATCH   /api/v1/students/:id                        # Update student information (Admin, Moderator)
DELETE  /api/v1/students/:id                        # Permanently delete a student (Admin)

GET     /api/v1/students/pending                    # List pending student registrations (Admin, Moderator)
GET     /api/v1/students/pending/:id                # Get pending registration details (Admin, Moderator)
DELETE  /api/v1/students/pending/:id                # Reject/delete pending registration (Admin, Moderator)
PATCH   /api/v1/students/pending/:id/verify         # Approve pending registration (Admin, Moderator)


======================================================
4. MODERATOR MANAGEMENT API
======================================================
GET     /api/v1/moderators                          # List moderators (compact payload) (Admin)
GET     /api/v1/moderators/:id                      # Get moderator details (Admin)
POST    /api/v1/moderators                          # Create a moderator account (Admin)
DELETE  /api/v1/moderators/:id                      # Delete a moderator (Admin)


======================================================
5. PORTFOLIO MANAGEMENT API
======================================================
GET     /api/v1/portfolio                           # Get portfolio content (Public)
PUT     /api/v1/portfolio                           # Replace portfolio content (Admin)


======================================================
6. ARTICLE MANAGEMENT API
======================================================
GET     /api/v1/articles                            # List articles (filter by status=draft|active|inactive) (Admin, Moderator)
GET     /api/v1/articles/:id                        # Get article details (Admin, Moderator)
POST    /api/v1/articles                            # Create an article (Moderator articles start as DRAFT)
PATCH   /api/v1/articles/:id                        # Update an article (Moderator can edit own drafts only)
PATCH   /api/v1/articles/:id/status                 # Change article status (Admin)
DELETE  /api/v1/articles/:id                        # Permanently delete an article (Admin)

GET     /api/v1/articles/public                     # List published articles (Public)
GET     /api/v1/articles/public/:id                 # Get published article details (Public)


======================================================
7. COLLABORATOR MANAGEMENT API
======================================================
GET     /api/v1/collaborators                       # List collaborators (Public)
POST    /api/v1/collaborators                       # Create collaborator (Admin, Moderator)
PATCH   /api/v1/collaborators/:id                   # Update collaborator (Admin, Moderator)
DELETE  /api/v1/collaborators/:id                   # Delete collaborator (Admin)


======================================================
8. CONTACT API
======================================================
GET     /api/v1/contact/meta                        # Get public contact information (Public)

POST    /api/v1/contact                             # Submit a contact message (Public)
GET     /api/v1/contact                             # List received messages (compact payload) (Admin)
GET     /api/v1/contact/:id                         # Get message details (Admin)
DELETE  /api/v1/contact/:id                         # Delete a message (Admin)


======================================================
9. RESEARCH REPOSITORY API
======================================================
GET     /api/v1/repository/documents                # List repository documents (Student, Moderator, Admin)
GET     /api/v1/repository/documents/:id            # Get repository document details/link (Student, Moderator, Admin)
POST    /api/v1/repository/documents                # Add a repository document (Admin)
PATCH   /api/v1/repository/documents/:id            # Update repository document (Moderator, Admin)
DELETE  /api/v1/repository/documents/:id            # Delete repository document (Admin)


======================================================
